cmake_minimum_required(VERSION 3.20)
project(EliteGE VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Vulkan REQUIRED)
find_program(GLSLC glslc HINTS "$ENV{VULKAN_SDK}/Bin" "$ENV{VULKAN_SDK}/bin" REQUIRED)
set(STB_HEADER "${CMAKE_CURRENT_BINARY_DIR}/stb_image.h")
set(STB_URL "https://raw.githubusercontent.com/nothings/stb/6b5799bae8a1fe3792691ef4b72ae26434c3e799/stb_image.h")
file(DOWNLOAD "${STB_URL}" "${STB_HEADER}"
    EXPECTED_HASH SHA256=dfc2aadda6295f67f677587773f25e5bf8c8ad427ed0db7c518525c91e72e020
    STATUS STB_DOWNLOAD_STATUS TLS_VERIFY ON)
list(GET STB_DOWNLOAD_STATUS 0 STB_DOWNLOAD_CODE)
list(GET STB_DOWNLOAD_STATUS 1 STB_DOWNLOAD_MESSAGE)
if(NOT STB_DOWNLOAD_CODE EQUAL 0)
    message(FATAL_ERROR "Failed to download verified stb_image.h: ${STB_DOWNLOAD_MESSAGE}")
endif()
set(VMA_HEADER "${CMAKE_CURRENT_BINARY_DIR}/vk_mem_alloc.h")
set(VMA_URL "https://raw.githubusercontent.com/GPUOpen-LibrariesAndSDKs/VulkanMemoryAllocator/v3.1.0/include/vk_mem_alloc.h")
file(DOWNLOAD "${VMA_URL}" "${VMA_HEADER}"
    EXPECTED_HASH SHA256=f063056705f2cd0ddd8d9da034f673b09814ca96aa45a958e9e687bf9f3fed15
    STATUS VMA_DOWNLOAD_STATUS TLS_VERIFY ON)
list(GET VMA_DOWNLOAD_STATUS 0 VMA_DOWNLOAD_CODE)
list(GET VMA_DOWNLOAD_STATUS 1 VMA_DOWNLOAD_MESSAGE)
if(NOT VMA_DOWNLOAD_CODE EQUAL 0)
    message(FATAL_ERROR "Failed to download verified vk_mem_alloc.h: ${VMA_DOWNLOAD_MESSAGE}")
endif()
include(FetchContent)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.3.8 GIT_SHALLOW TRUE)
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.1 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(glfw glm)
set(SHADER_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/shaders")
file(MAKE_DIRECTORY "${SHADER_OUTPUT_DIR}")
add_custom_command(OUTPUT "${SHADER_OUTPUT_DIR}/quad.vert.spv"
    COMMAND "${GLSLC}" -DVERTEX_SHADER -fshader-stage=vert "${CMAKE_CURRENT_SOURCE_DIR}/shaders/quad.glsl" -o "${SHADER_OUTPUT_DIR}/quad.vert.spv"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/quad.glsl")
add_custom_command(OUTPUT "${SHADER_OUTPUT_DIR}/quad.frag.spv"
    COMMAND "${GLSLC}" -fshader-stage=frag "${CMAKE_CURRENT_SOURCE_DIR}/shaders/quad.glsl" -o "${SHADER_OUTPUT_DIR}/quad.frag.spv"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/quad.glsl")
add_custom_target(EliteGEShaders DEPENDS "${SHADER_OUTPUT_DIR}/quad.vert.spv" "${SHADER_OUTPUT_DIR}/quad.frag.spv")
add_executable(EliteGE main.cpp)
add_dependencies(EliteGE EliteGEShaders)
target_compile_definitions(EliteGE PRIVATE GLFW_INCLUDE_NONE ELITEGE_SHADER_DIR="${SHADER_OUTPUT_DIR}" ELITEGE_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
target_include_directories(EliteGE SYSTEM PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(EliteGE PRIVATE Vulkan::Vulkan glfw glm::glm)
if(MSVC)
    target_compile_options(EliteGE PRIVATE /W4)
else()
    target_compile_options(EliteGE PRIVATE -Wall -Wextra -Wpedantic)
endif()
